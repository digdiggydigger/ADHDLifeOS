import SwiftUI
import XCTest
@testable import ADHD_LifeOS

@MainActor
final class AuditSprintControlsRenderProbe: XCTestCase {

    private static let outDir = "/private/tmp/claude-501/-Users-ethan--E--Claude-Code-v1-GoogleAI-ADHDLifeOS-ADHDLifeOS/583f35f8-93dd-4815-8059-9c715c0409e1/scratchpad/r4-ctl"

    func testRenderSprintControlOptions() throws {
        try FileManager.default.createDirectory(atPath: Self.outDir, withIntermediateDirectories: true)
        _ = ImageRenderer(content: Text("warm-up")).uiImage
        var count = 0
        for scheme in [ColorScheme.light, .dark] {
            let s = scheme == .dark ? "D" : "L"
            for size in [DynamicTypeSize.large, .accessibility3] {
                let z = size == .large ? "std" : "AX3"
                for layout in CardLayout.allCases {
                    count += try render(MockCard(layout: layout), name: "card-\(layout.rawValue)-\(s)-\(z)",
                                        scheme: scheme, size: size, height: nil)
                }
            }
            // Today's real card, for reference.
            let session = FocusSession(taskId: nil, taskTitle: "Pay the council tax instalment", lifeAreaEmoji: "💰",
                                       durationSeconds: 25 * 60, remainingSeconds: 14 * 60 + 57,
                                       nudgeCheckpoints: [300, 900])
            let real = FocusTimerBarContent(session: session, isCollapsed: false,
                                            checkpointBanner: nil, reduceMotion: true,
                                            onToggleCollapse: {}, onTogglePause: {}, onExtend: { _ in }, onRequestStop: {})
                .padding(.horizontal, 16)
                .padding(.vertical, 16)
                .background(Color.cardSurface, in: RoundedRectangle(cornerRadius: FocusBarMetrics.cornerRadius, style: .continuous))
                .overlay(RoundedRectangle(cornerRadius: FocusBarMetrics.cornerRadius, style: .continuous)
                    .strokeBorder(Color.accentColor.opacity(0.3), lineWidth: 1))
                .padding(.horizontal, 16)
            count += try render(real, name: "card-today-\(s)-std", scheme: scheme, size: .large, height: nil)
            for custom in [false, true] {
                count += try render(MockFocusScreen(showsCustom: custom), name: "screen-\(custom ? "custom" : "plain")-\(s)-std",
                                    scheme: scheme, size: .large, height: 812)
            }
            count += try render(MockFocusScreen(showsCustom: false), name: "screen-plain-\(s)-AX3",
                                scheme: scheme, size: .accessibility3, height: 812)
        }
        XCTAssertEqual(count, 2 * (2 * CardLayout.allCases.count + 1 + 3))
    }

    private func render<V: View>(_ view: V, name: String, scheme: ColorScheme, size: DynamicTypeSize,
                                 height: CGFloat?) throws -> Int {
        let framed = view
            .environment(\.colorScheme, scheme)
            .dynamicTypeSize(size)
            .frame(width: 402, height: height)
            .padding(.vertical, height == nil ? 16 : 0)
        let renderer = ImageRenderer(content: framed)
        renderer.scale = 3
        renderer.isOpaque = false
        var attempt = renderer.uiImage
        for _ in 0..<3 where attempt == nil {
            RunLoop.current.run(until: Date().addingTimeInterval(0.2))
            attempt = renderer.uiImage
        }
        guard let image = attempt, let png = image.pngData() else {
            XCTFail("nothing rendered for \(name)")
            return 0
        }
        XCTAssertGreaterThan(image.size.height, 40, name)
        try png.write(to: URL(fileURLWithPath: "\(Self.outDir)/\(name).png"))
        return 1
    }
}

// MARK: - Controls (house tokens only)

private enum SprintControl: CaseIterable {
    case pause, add30, add60, add300, custom, end

    var glyph: String {
        switch self {
        case .pause: return "pause.fill"
        case .add30: return "goforward.30"
        case .add60, .add300: return "goforward.plus"
        case .custom: return "timer"
        case .end: return "stop.fill"
        }
    }
    var title: String {
        switch self {
        case .pause: return "Pause"
        case .add30: return "+30 sec"
        case .add60: return "+1 min"
        case .add300: return "+5 min"
        case .custom: return "Custom"
        case .end: return "End"
        }
    }
    var big: String {
        switch self {
        case .add30: return "+30"
        case .add60: return "+1"
        case .add300: return "+5"
        default: return ""
        }
    }
    var unit: String {
        switch self {
        case .add30: return "sec"
        case .add60, .add300: return "min"
        default: return title
        }
    }
    var tint: Color { self == .end ? Color("StateRisk") : .accentColor }
}

/// A tinted capsule, the same language as the chosen undo capsule (round 2b).
private struct Tinted: ViewModifier {
    let tint: Color
    @Environment(\.colorScheme) private var scheme
    func body(content: Content) -> some View {
        content
            .foregroundStyle(tint)
            .background(
                tint.opacity(scheme == .dark ? AppTabBarMetrics.chipTintDark : AppTabBarMetrics.chipTintLight),
                in: RoundedRectangle(cornerRadius: 16, style: .continuous)
            )
    }
}

/// Glyph (or big number) over a small label: fits six across 338pt at 48pt wide.
private struct StackedButton: View {
    let control: SprintControl
    var body: some View {
        VStack(spacing: 4) {
            if control.big.isEmpty {
                Image(systemName: control.glyph).font(.body.weight(.semibold))
            } else {
                Text(control.big).font(.headline.monospacedDigit())
            }
            Text(control.unit).font(.caption2.weight(.semibold))
        }
        .frame(maxWidth: .infinity, minHeight: 56)
        .modifier(Tinted(tint: control.tint))
    }
}

private struct WideButton: View {
    let control: SprintControl
    var height: CGFloat = 48
    private var showsGlyph: Bool { control == .pause || control == .end }
    var body: some View {
        Group {
            if showsGlyph {
                Label(control.title, systemImage: control.glyph)
            } else {
                Text(control.title)
            }
        }
            .font(.subheadline.weight(.semibold))
            .lineLimit(2)
            .multilineTextAlignment(.center)
            .padding(.horizontal, 8)
            .frame(maxWidth: .infinity, minHeight: height)
            .modifier(Tinted(tint: control.tint))
    }
}

// MARK: - The card

private enum CardLayout: String, CaseIterable {
    case oneRow = "V1-one-row"
    case twoRows = "V2-two-rows"
    case timeMenu = "V3-time-menu"
}

private struct MockRing: View {
    var diameter: CGFloat = 70
    var lineWidth: CGFloat = 6
    var timeFont: Font = .callout.monospacedDigit().weight(.semibold)
    var body: some View {
        ZStack {
            Circle().stroke(Color("LabelTertiary").opacity(0.5), lineWidth: lineWidth)
            Circle().trim(from: 0, to: 0.4).stroke(Color.accentColor, style: StrokeStyle(lineWidth: lineWidth, lineCap: .round))
                .rotationEffect(.degrees(-90))
            Text("14:57").font(timeFont).minimumScaleFactor(0.4).lineLimit(1).padding(.horizontal, 8)
        }
        .frame(width: diameter, height: diameter)
    }
}

private struct MockCard: View {
    let layout: CardLayout
    @Environment(\.dynamicTypeSize) private var typeSize

    var body: some View {
        VStack(alignment: .leading, spacing: 16) {
            if typeSize.isAccessibilitySize {
                MockRing(diameter: 128, lineWidth: 8)
                title
            } else {
                HStack(spacing: 16) {
                    MockRing()
                    title
                    Spacer(minLength: 0)
                }
            }
            controls
        }
        .padding(16)
        .background(Color.cardSurface, in: RoundedRectangle(cornerRadius: FocusBarMetrics.cornerRadius, style: .continuous))
        .overlay(RoundedRectangle(cornerRadius: FocusBarMetrics.cornerRadius, style: .continuous)
            .strokeBorder(Color.accentColor.opacity(0.3), lineWidth: 1))
        .shadow(color: Color.black.opacity(0.03), radius: 10, x: 0, y: 4)
        .padding(.horizontal, 16)
    }

    private var title: some View {
        HStack(alignment: .firstTextBaseline, spacing: 8) {
            Text("💰")
            Text("Pay the council tax instalment")
                .font(.subheadline.weight(.semibold))
                .lineLimit(typeSize.isAccessibilitySize ? 4 : 2)
        }
    }

    @ViewBuilder
    private var controls: some View {
        let ax = typeSize.isAccessibilitySize
        switch layout {
        case .oneRow:
            if ax {
                VStack(spacing: 8) {
                    HStack(spacing: 8) { ForEach([SprintControl.pause, .add30, .add60], id: \.self) { StackedButton(control: $0) } }
                    HStack(spacing: 8) { ForEach([SprintControl.add300, .custom, .end], id: \.self) { StackedButton(control: $0) } }
                }
            } else {
                HStack(spacing: 8) { ForEach(SprintControl.allCases, id: \.self) { StackedButton(control: $0) } }
            }
        case .twoRows:
            VStack(spacing: 8) {
                if ax {
                    HStack(spacing: 8) { WideButton(control: .add30); WideButton(control: .add60) }
                    HStack(spacing: 8) { WideButton(control: .add300); WideButton(control: .custom) }
                } else {
                    HStack(spacing: 8) { ForEach([SprintControl.add30, .add60, .add300, .custom], id: \.self) { WideButton(control: $0) } }
                }
                HStack(spacing: 8) { WideButton(control: .pause); WideButton(control: .end) }
            }
        case .timeMenu:
            let add = Label("Add time", systemImage: "plus.circle.fill")
                .font(.subheadline.weight(.semibold))
                .frame(maxWidth: .infinity, minHeight: 48)
                .modifier(Tinted(tint: .accentColor))
            if ax {
                VStack(spacing: 8) { WideButton(control: .pause); add; WideButton(control: .end) }
            } else {
                HStack(spacing: 8) { WideButton(control: .pause); add; WideButton(control: .end) }
            }
        }
    }
}

/// Top-rounded sheet shape without needing iOS 17's UnevenRoundedRectangle.
private struct UnevenRoundedRectangleCompat: Shape {
    func path(in rect: CGRect) -> Path {
        let r: CGFloat = 38
        var p = Path()
        p.move(to: CGPoint(x: 0, y: rect.maxY))
        p.addLine(to: CGPoint(x: 0, y: r))
        p.addQuadCurve(to: CGPoint(x: r, y: 0), control: .zero)
        p.addLine(to: CGPoint(x: rect.maxX - r, y: 0))
        p.addQuadCurve(to: CGPoint(x: rect.maxX, y: r), control: CGPoint(x: rect.maxX, y: 0))
        p.addLine(to: CGPoint(x: rect.maxX, y: rect.maxY))
        p.closeSubpath()
        return p
    }
}
